const { onCall, onRequest, HttpsError } = require("firebase-functions/v2/https"); const { defineSecret } = require("firebase-functions/params"); const crypto = require("crypto"); const admin = require("firebase-admin"); admin.initializeApp(); const RESEND_API_KEY = defineSecret("RESEND_API_KEY"); const PROJECT_REGION = "us-central1"; const PROJECT_ID = "maths-mate"; const FUNCTIONS_BASE_URL = `https://${PROJECT_REGION}-${PROJECT_ID}.cloudfunctions.net`; /** * Sends one email via Resend. Every call is wrapped in try/catch by its caller — a * failed notification email should never block the actual request/approval/denial from * going through, since the in-app Pending Requests screen always works as a fallback * regardless of whether email delivery succeeds. */ async function sendEmail(apiKey, { to, subject, html }) { const res = await fetch("https://api.resend.com/emails", { method: "POST", headers: { "Authorization": `Bearer ${apiKey}`, "Content-Type": "application/json", }, body: JSON.stringify({ from: "Maths Mate ", to: [to], subject, html, }), }); if (!res.ok) { const body = await res.text(); throw new Error(`Resend API error ${res.status}: ${body}`); } } function approveDenyEmailHtml(name, email, approveUrl, denyUrl) { return `

New teacher request

${name} (${email}) wants teacher access to Maths Mate.

Approve Deny

You can also review this anytime in the app under Students → Pending requests.

`; } function simpleHtmlPage(title, message) { return `${title}

${title}

${message}

`; } /** * Confirms the caller is a signed-in teacher, and that the target student genuinely * belongs to that teacher's class. Shared by every function below that acts on a * specific student, so this ownership check lives in exactly one place rather than * being copied (and potentially drifting) across each function. */ async function requireOwnedStudent(request) { if (!request.auth) { throw new HttpsError("unauthenticated", "You must be signed in."); } const callerUid = request.auth.uid; const targetUid = request.data && request.data.uid; if (!targetUid || typeof targetUid !== "string") { throw new HttpsError("invalid-argument", "Missing student uid."); } const db = admin.firestore(); const callerRecord = await admin.auth().getUser(callerUid); const teachersDoc = await db.collection("config").doc("teachers").get(); const teacherEmails = (teachersDoc.exists && teachersDoc.data().emails) || []; const callerEmail = (callerRecord.email || "").toLowerCase(); const isTeacher = teacherEmails.map((e) => String(e).toLowerCase()).includes(callerEmail); if (!isTeacher) { throw new HttpsError("permission-denied", "Only teachers can do this."); } const studentSnap = await db.collection("profiles").doc(targetUid).get(); if (!studentSnap.exists) { throw new HttpsError("not-found", "Student not found."); } const student = studentSnap.data(); if (student.teacherUid !== callerUid) { throw new HttpsError("permission-denied", "This student is not in your class."); } return { db, callerUid, targetUid, student }; } /** * Resets a student's password. Only callable by a signed-in teacher, and only for a * student that actually belongs to that teacher's class (checked via the student's * profile.teacherUid). Generates a fresh random password, sets it on the student's * Firebase Auth account (their existing account, uid, and all their history stay * exactly the same — this is a real reset, not a new account), saves a copy into * their profile so the teacher can see it on the Students page, and returns it. * * Written with the explicit firebase-functions/v2 API (rather than the older * functions.https.onCall style) so "who is calling this" is read unambiguously — * mixing the old and new styles on a 2nd-generation function is what caused the * earlier 401 "unauthenticated" errors even when the caller genuinely was signed in. */ exports.resetStudentPassword = onCall(async (request) => { const { db, targetUid, student } = await requireOwnedStudent(request); const words = [ "tiger", "comet", "rocket", "biscuit", "dolphin", "cactus", "maple", "otter", "pretzel", "falcon", "koala", "muffin", "panda", "turtle", "walnut", "banjo", "cobra", "gizmo", "jungle", "lemon", "mango", "nimbus", "pickle", "quokka", ]; const word = words[Math.floor(Math.random() * words.length)]; const num = 10 + Math.floor(Math.random() * 90); const newPassword = `${word}${num}`; await admin.auth().updateUser(targetUid, { password: newPassword }); await db.collection("profiles").doc(targetUid).update({ currentPassword: newPassword }); return { password: newPassword, username: student.username || null, displayName: student.displayName || null, }; }); /** * Permanently deletes a student: their Firebase Auth login, their profile, and every * record of their work (weekly quiz results, timetable attempts, skill practice * sessions, saved in-progress quizzes). This is irreversible — the client-side * confirmation makes the teacher type the student's exact name before calling this, * but this function does not re-check that itself (it can't see what was typed) — * it trusts that a call here means "yes, really delete." */ exports.deleteStudent = onCall(async (request) => { const { db, targetUid, student } = await requireOwnedStudent(request); // Clean up every collection that stores records keyed by this student's uid. // Firestore doesn't cascade-delete, so each has to be found and removed explicitly. const collectionsToClean = ["results", "ttAttempts", "skillPractice", "weekProgress"]; for (const coll of collectionsToClean) { const snap = await db.collection(coll).where("uid", "==", targetUid).get(); if (!snap.empty) { const batch = db.batch(); snap.docs.forEach((d) => batch.delete(d.ref)); await batch.commit(); } } await db.collection("profiles").doc(targetUid).delete(); await admin.auth().deleteUser(targetUid); return { deleted: true, displayName: student.displayName || null }; }); /** * Reads config/teachers and works out both the full teacher list and who's an admin. * If no "admins" field has been set up yet, every current teacher is treated as an * admin (preserves existing behavior until a real admin list is configured) — see * the one-time setup note in the invite/revoke functions below. */ async function getTeacherConfig(db) { const doc = await db.collection("config").doc("teachers").get(); const data = (doc.exists && doc.data()) || {}; const emails = (data.emails || []).map((e) => String(e).toLowerCase()); const admins = data.admins ? data.admins.map((e) => String(e).toLowerCase()) : null; return { emails, admins }; } /** * Invites a colleague as a new, fully independent teacher. Only an ADMIN teacher can * call this — a plain (non-admin) teacher who was themselves invited cannot invite * further colleagues, so access can't silently spread beyond who the admin approves. * Creates their Auth account (with a throwaway password — they set their own via the * standard Firebase password-reset email, sent separately by the client after this * succeeds, since teachers have real addresses this can actually reach) and adds their * email to config/teachers.emails — the ONLY way that list can ever be written to, * since client-side rules block writing to it entirely (see firestore.rules). * * Deliberately does NOT copy or link any data from the inviting teacher — every teacher's * students, classes, and revision sets are already isolated by teacherUid throughout the * whole app, so a newly invited teacher starts with a completely empty, private space. */ /** * Actually creates a teacher's account and grants them access — used both by a direct * admin invite and by approving a self-submitted request, so there's exactly one place * that does this rather than two copies that could drift apart. Also writes a * teacherDirectory entry (name + email + status), which is what makes it possible to * later browse "everyone who's ever had access" with real names, not just a bare list * of emails inside config/teachers. */ async function provisionTeacherAccount(db, email, name, referredBy) { const tempPassword = Math.random().toString(36).slice(2) + Math.random().toString(36).slice(2) + "!Aa1"; let userRecord; try { userRecord = await admin.auth().createUser({ email, password: tempPassword, displayName: name || email }); } catch (e) { if (e.code === "auth/email-already-exists") { userRecord = await admin.auth().getUserByEmail(email); } else { throw new HttpsError("internal", "Could not create the account: " + e.message); } } await db.collection("config").doc("teachers").set( { emails: admin.firestore.FieldValue.arrayUnion(email) }, { merge: true } ); await db.collection("teacherDirectory").doc(userRecord.uid).set({ name: name || email, email, status: "active", approvedAt: admin.firestore.FieldValue.serverTimestamp(), referredBy: referredBy || null, }); return userRecord; } /** * Confirms the caller is an admin teacher (per config/teachers.admins, falling back to * "any current teacher" if that field hasn't been configured yet). Shared by every * admin-only function below. */ async function requireAdminTeacher(request) { if (!request.auth) { throw new HttpsError("unauthenticated", "You must be signed in."); } const db = admin.firestore(); const callerRecord = await admin.auth().getUser(request.auth.uid); const callerEmail = (callerRecord.email || "").toLowerCase(); const { emails: teacherEmails, admins } = await getTeacherConfig(db); const isAdmin = admins ? admins.includes(callerEmail) : teacherEmails.includes(callerEmail); if (!isAdmin) { throw new HttpsError("permission-denied", "Only an admin teacher can do this."); } return { db, callerEmail, teacherEmails }; } exports.inviteTeacher = onCall(async (request) => { const { db, teacherEmails } = await requireAdminTeacher(request); const name = request.data && request.data.name; const emailRaw = request.data && request.data.email; if (!emailRaw || typeof emailRaw !== "string" || !emailRaw.includes("@")) { throw new HttpsError("invalid-argument", "A valid email address is required."); } const email = emailRaw.trim().toLowerCase(); if (teacherEmails.includes(email)) { throw new HttpsError("already-exists", "That email is already a teacher on this account."); } const userRecord = await provisionTeacherAccount(db, email, name); return { uid: userRecord.uid, email }; }); /** * Revokes a teacher's access. Only an admin can call this, and an admin can't revoke * themselves (to avoid ever locking every admin out at once). Removes the email from * config/teachers.emails immediately — since every security rule re-checks that list * live on each request rather than trusting anything cached in their sign-in token, * this cuts off their ability to read or write any teacher-scoped data right away — and * additionally disables their Firebase Auth account outright, so they can't sign in * again at all, not just lose teacher permissions. Also marks their teacherDirectory * entry as revoked (rather than deleting it), so the full history — including who was * once here and later cut off — stays visible for your own records. */ exports.revokeTeacher = onCall(async (request) => { const { db, callerEmail } = await requireAdminTeacher(request); const emailRaw = request.data && request.data.email; if (!emailRaw || typeof emailRaw !== "string") { throw new HttpsError("invalid-argument", "Missing email."); } const email = emailRaw.trim().toLowerCase(); if (email === callerEmail) { throw new HttpsError("invalid-argument", "You can't revoke your own access."); } await db.collection("config").doc("teachers").update({ emails: admin.firestore.FieldValue.arrayRemove(email), }); try { const targetUser = await admin.auth().getUserByEmail(email); await admin.auth().updateUser(targetUser.uid, { disabled: true }); await db.collection("teacherDirectory").doc(targetUser.uid).set( { status: "revoked", revokedAt: admin.firestore.FieldValue.serverTimestamp() }, { merge: true } ); } catch (e) { // If we couldn't find/disable their Auth account, removing them from the teachers // list above still fully blocks their access — this is a best-effort extra step. } return { revoked: true, email }; }); /** * Public — anyone can call this, signed in or not. Records a request to become a * teacher; does NOT create any account or grant any access on its own. This is the * self-serve front door: a teacher who finds the app can ask for access without you * having to know about them first, but nothing is provisioned until you approve it. * Also emails every admin a one-tap Approve/Deny link, so a request can be handled * from a phone or watch notification without opening the app at all. */ exports.requestTeacherAccess = onCall({ secrets: [RESEND_API_KEY] }, async (request) => { const db = admin.firestore(); const name = request.data && request.data.name; const emailRaw = request.data && request.data.email; const referredByRaw = request.data && request.data.referredBy; if (!name || typeof name !== "string" || !name.trim()) { throw new HttpsError("invalid-argument", "A name is required."); } if (!emailRaw || typeof emailRaw !== "string" || !emailRaw.includes("@")) { throw new HttpsError("invalid-argument", "A valid email address is required."); } const email = emailRaw.trim().toLowerCase(); const referredBy = (referredByRaw && typeof referredByRaw === "string") ? referredByRaw.trim().toLowerCase() : null; const { emails: teacherEmails, admins } = await getTeacherConfig(db); if (teacherEmails.includes(email)) { throw new HttpsError("already-exists", "This email already has teacher access."); } const existingPending = await db.collection("teacherRequests") .where("email", "==", email) .where("status", "==", "pending") .limit(1) .get(); if (!existingPending.empty) { return { submitted: true, alreadyPending: true }; } const approvalToken = crypto.randomBytes(24).toString("hex"); const reqRef = await db.collection("teacherRequests").add({ name: name.trim(), email, status: "pending", requestedAt: admin.firestore.FieldValue.serverTimestamp(), referredBy, approvalToken, }); try { const notifyList = admins || teacherEmails; const approveUrl = `${FUNCTIONS_BASE_URL}/approveByEmail?requestId=${reqRef.id}&token=${approvalToken}`; const denyUrl = `${FUNCTIONS_BASE_URL}/denyByEmail?requestId=${reqRef.id}&token=${approvalToken}`; const html = approveDenyEmailHtml(name.trim(), email, approveUrl, denyUrl); for (const adminEmail of notifyList) { await sendEmail(RESEND_API_KEY.value(), { to: adminEmail, subject: `New teacher request: ${name.trim()}`, html }); } } catch (e) { // A failed notification email should never block the request itself — it still // shows up in the in-app Pending Requests list as a fallback either way. console.warn("Could not send admin notification email:", e); } return { submitted: true, alreadyPending: false }; }); /** * Admin-only. Turns a pending request into an actual teacher account via the same * provisioning path a direct invite uses, then marks the request approved. This is * the in-app version of approval (used by the Pending Requests screen) — the emailed * one-tap link uses approveByEmail below, which does the same core work without * requiring the admin to be signed into the app at all. */ exports.approveTeacherRequest = onCall(async (request) => { const { db } = await requireAdminTeacher(request); const requestId = request.data && request.data.requestId; if (!requestId || typeof requestId !== "string") { throw new HttpsError("invalid-argument", "Missing requestId."); } const reqRef = db.collection("teacherRequests").doc(requestId); const reqSnap = await reqRef.get(); if (!reqSnap.exists) { throw new HttpsError("not-found", "That request no longer exists."); } const reqData = reqSnap.data(); if (reqData.status !== "pending") { throw new HttpsError("failed-precondition", "That request has already been decided."); } const userRecord = await provisionTeacherAccount(db, reqData.email, reqData.name, reqData.referredBy); await reqRef.update({ status: "approved", decidedAt: admin.firestore.FieldValue.serverTimestamp(), }); return { uid: userRecord.uid, email: reqData.email }; }); /** * The one-tap "Approve" link from the admin notification email. No sign-in required — * the random approvalToken generated when the request was created IS the authentication, * so this only works for someone who actually has that exact email in hand. Provisions * the account (same shared path as everywhere else), then emails the new teacher their * own password-setup link directly, since there's no signed-in browser session here to * fall back on the usual client-side Firebase call. */ exports.approveByEmail = onRequest({ secrets: [RESEND_API_KEY] }, async (req, res) => { const { requestId, token } = req.query; if (!requestId || !token) { res.status(400).send(simpleHtmlPage("Missing link details", "This approval link looks incomplete.")); return; } const db = admin.firestore(); const reqRef = db.collection("teacherRequests").doc(String(requestId)); const reqSnap = await reqRef.get(); if (!reqSnap.exists) { res.status(404).send(simpleHtmlPage("Not found", "This request no longer exists.")); return; } const reqData = reqSnap.data(); if (reqData.approvalToken !== token) { res.status(403).send(simpleHtmlPage("Link not valid", "This approval link doesn't match our records.")); return; } if (reqData.status !== "pending") { res.status(200).send(simpleHtmlPage("Already decided", `This request was already marked "${reqData.status}".`)); return; } const userRecord = await provisionTeacherAccount(db, reqData.email, reqData.name, reqData.referredBy); await reqRef.update({ status: "approved", decidedAt: admin.firestore.FieldValue.serverTimestamp() }); try { const setupLink = await admin.auth().generatePasswordResetLink(reqData.email); await sendEmail(RESEND_API_KEY.value(), { to: reqData.email, subject: "You're in! Set up your Maths Mate password", html: `

Hi ${reqData.name},

You've been approved as a teacher on Maths Mate. Click below to set your password and get started:

Set your password

`, }); } catch (e) { console.warn("Could not send password-setup email:", e); } res.status(200).send(simpleHtmlPage("Approved ✓", `${reqData.name} now has teacher access, and they've been emailed a link to set up their password.`)); }); /** * The one-tap "Deny" link from the admin notification email — same token-based * authentication as approveByEmail, no sign-in required. */ exports.denyByEmail = onRequest(async (req, res) => { const { requestId, token } = req.query; if (!requestId || !token) { res.status(400).send(simpleHtmlPage("Missing link details", "This link looks incomplete.")); return; } const db = admin.firestore(); const reqRef = db.collection("teacherRequests").doc(String(requestId)); const reqSnap = await reqRef.get(); if (!reqSnap.exists) { res.status(404).send(simpleHtmlPage("Not found", "This request no longer exists.")); return; } const reqData = reqSnap.data(); if (reqData.approvalToken !== token) { res.status(403).send(simpleHtmlPage("Link not valid", "This link doesn't match our records.")); return; } if (reqData.status !== "pending") { res.status(200).send(simpleHtmlPage("Already decided", `This request was already marked "${reqData.status}".`)); return; } await reqRef.update({ status: "denied", decidedAt: admin.firestore.FieldValue.serverTimestamp() }); res.status(200).send(simpleHtmlPage("Denied", `${reqData.name}'s request was denied. No account was created.`)); }); /** * Admin-only. Marks a pending request as denied — no account is ever created for it. * Kept (not deleted) so there's a record of who asked and what was decided. */ exports.denyTeacherRequest = onCall(async (request) => { const { db } = await requireAdminTeacher(request); const requestId = request.data && request.data.requestId; if (!requestId || typeof requestId !== "string") { throw new HttpsError("invalid-argument", "Missing requestId."); } const reqRef = db.collection("teacherRequests").doc(requestId); const reqSnap = await reqRef.get(); if (!reqSnap.exists) { throw new HttpsError("not-found", "That request no longer exists."); } if (reqSnap.data().status !== "pending") { throw new HttpsError("failed-precondition", "That request has already been decided."); } await reqRef.update({ status: "denied", decidedAt: admin.firestore.FieldValue.serverTimestamp(), }); return { denied: true }; });
